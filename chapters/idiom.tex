\section{按话题分}

\subsection{\textcolor{yellow}{die} Prüfung}
\noindent
A: Ich mache morgen die Prüfung. \marginpar{die Prüfung \square[color=yellow]: exam} \\
B: Viel Erfolg! \marginpar{der Erfolg \square[color=green]: success} \\

\subsection{\textcolor{yellow}{die} Verspätung}
\noindent
A: Ich komme zu spät. Entschuldigung. \marginpar{normal apology} \\
A: (Es) tut mir leid. \marginpar{deeper apology} \\

\subsection{\textcolor{green}{der} Geburstag}
\noindent
B: Herzlichen Glückwunsch zum Geburstag! \marginpar{此处省略Ich wünsche dir 所以后面是宾格 \\ der Glückwunsch \square[color=green] (zu): good wishes (to)} \\
\\
\noindent
B: Alles Gute zum Geburstag! \marginpar{das Gute \square[color=purple]: good thing, decl. as adj.} \\
\\
\noindent
B: Viel Erfolg und vor allem Gesundheit im neuen Lebensjahr! \marginpar{vor allem: above all \\ gesund \square[border=true, scale=0.8]: healthy \\ die Gesundheit \square[color=yellow]: health \\ das Lebensjahr \square[color=purple]: age} \\
\\
\\
\\
\noindent

\subsection{\textcolor{yellow}{die} Geburt}
A: Wir haben einen Sohn / eine Tochter! \\
B: Herzlichen Glückwunsch zu eurem Sohn / eurer Tochter! \\
B: Wir freuen uns sehr für euch! \\

\subsection{\textcolor{yellow}{die} Krankheit}
\noindent
A: Ich bin krank. \\
B: Gute Besserung! \\
B: Alles Gute! \marginpar{alles 代词相当于冠词角色} \\

\subsection{\textcolor{yellow}{die} Hochzeit}
A: Alles Gute zur Hochzeit! \\
A: Viel Glück für euch beide! \marginpar{das Glück \square[color=purple]: luck \\ beide 此处是数量代词做euch宾补} \\

\subsection{Verabschieden}

B: Komm gut nach Hause! \\
B: Gute Fahrt! \\

B: Viel Spaß! \marginpar{der Spaß \square[color=green]: fun, nur Sg. \\ Spaß haben: have fun \\ Spaß machen: bring fun} \\

\subsection{Niesen}

\section{按词形分}
\subsection{all- ...}
\begin{itemize}
    \item Alles Gute!
    \item Alles Klar!
    \item Alles in Ordnung
\end{itemize}

\subsection{viel ...} 
\begin{itemize}
    \item Viel Erfolg! 
    \item Viel Glück!
    \item Viel Spaß!
\end{itemize}

\subsection{gut- ...}
\begin{itemize}
    \item Gute Besserung!
    \item Gute Reise!
    \item Gute Fahrt!
    \item Guten Appetit!
\end{itemize}

\subsection{kein- ...}
\begin{itemize}
    \item Kein Problem!
    \item Keine Sorge! \marginpar{die Sorge \square[color=yellow]: worry}
    \item Keine Angst! \marginpar{die Angst \square[color=yellow]: fear}
\end{itemize}

\subsection{总结}
\begin{itemize}
    \item viel不可数前是much，数量副词不变格
    \item alle, gut, viel 祝福语完整形式： \\
    Ich wunsche dir \textasciitilde.
    \item kein 完整形式大概：\\
    Es gibt kein- \textasciitilde
\end{itemize}
