% Styles
\tikzset{spelling/.style={
            inner sep=0pt,
            outer sep=0pt,
            node distance=0.4cm,
            anchor=center,
            font=\sffamily\bfseries,
        }}

\tikzset{verb/.style={
            node distance=0.2cm,
            minimum width=0.6cm,
        }}

% Flashcard
\pgfkeys{
    /flashcard/.is family, /flashcard,
    default/.style = {},
    color/.store in = \flashcardcolor,
    title/.store in = \flashcardtitle,
    picture/.store in = \flashcardpicture,
    hint/.store in = \flashcardhint,
    foot/.store in = \flashcardfoot,
    spelling/.store in = \flashcardspelling,
}

\NewDocumentCommand{\flashcard}{o}{%
    \pgfkeys{/flashcard, default, #1}%
    \begin{tikzpicture}%
        \pgfmathsetmacro{\w}{(5.71/1pt)}
        \pgfmathsetmacro{\h}{(\w/11*19)}
        \pgfmathsetmacro{\tb}{(\h*0.06)} % top bottom margin
        \pgfmathsetmacro{\lr}{(\w*0.025)}
        \pgfmathsetmacro{\ch}{(\h-\tb*2)}
        \pgfmathsetmacro{\cw}{(\w-\lr*2)}

        % frame
        \begin{scope}%
            \clip[overlay,even odd clip,reverseclip] (\lr, \tb) rectangle(\w-\lr, \h-\tb);%
            \draw[fill=\flashcardcolor, draw=none, rounded corners=\w*0.04*1cm] (0, 0) rectangle(\w, \h);%
        \end{scope}

        % title
        \expandafter\ifstrequal\expandafter{\flashcardcolor}{gray}{
            \def\titlecolor{black}
        }{
            \def\titlecolor{white}
        }
        %% use overlay so that it doesn't affect the bounding box
        \node[align=center, anchor=center, text=\titlecolor, font=\bfseries\sffamily, overlay] at (\w/2, \h - \tb/2) {\flashcardtitle};

        %% proportions: 10%, 30%, 50%, 10%
        % pronounciation
        \node[align=center, anchor=center, minimum height=\ch*0.1] at (\w/2, \ch*0.95+\tb) {
            % \begin{tikzpicture}
            %     \draw[fill=none, draw=black] (0,0) rectangle(\w-\lr*2,\ch*0.1);
            %     \node[align=center, anchor=center, text=white, overlay] at (\w/2, \ch*0.1/2) {pronounciation};
            % \end{tikzpicture}%
        };

        % spelling
        \node[spelling, minimum height=\ch*0.3] at (\w/2, \ch*0.75+\tb) {\flashcardspelling};

        % picture
        \node[minimum height=\ch*0.5, overlay] at (\w/2, \ch*0.35+\tb) {%
            \pgfmathsetmacro{\ih}{\ch*0.5}%
            \pgfmathsetmacro{\iw}{\cw*0.95}%
            \includegraphics[keepaspectratio, height=\ih cm, width=\iw cm]{\flashcardpicture}%
        };

        % hint
        \node[align=center, anchor=center, minimum height=\ch*0.1] at (\w/2, \ch*0.05+\tb) {
            \begin{tikzpicture}
                \node[align=center, anchor=center, text=gray] at (0, 0) {\flashcardhint};
            \end{tikzpicture}%
        };

        % footnote
        \node[align=center, anchor=center, text=\titlecolor, overlay] at (\w/2, \tb/2) {\flashcardfoot};

        \useasboundingbox (0,0) rectangle (\w, \h);

    \end{tikzpicture}%
}

\pgfkeys{
    /makeflashcard/.is family, /makeflashcard,
    default/.style = {translation={}, hint={}, tags={}, group={}},
    partofspeech/.store in = \fpart,
    % sound
    pronounciation/.store in = \fpronounciation,
    % form
    spelling/.store in = \fspelling,
    % meaning
    translation/.store in = \ftranslation,
    picture/.store in = \fpicture,
    hint/.store in = \fhint,
    tags/.store in = \ftags,
    group/.store in = \fgroups,
}

\NewDocumentCommand{\makeflashcard}{m}{
    \pgfkeys{/makeflashcard, default, #1}%

    \def\cardcolor{darkgray}

    \edef\p{\fpart}
    % noun
    \expandafter\ifstrequal\expandafter{\p}{n.}{
        % spelling
        \StrCount{\fspelling}{,}[\dc]
        \StrCut{\fspelling}{,}{\pa}{\pb}
        \StrCut{\pa}{ }{\article}{\cardtitle}
        \ifnum\dc=1
            \def\plural{\pb}
        \else
            \StrCut{\pb}{,}{\genitive}{\plural}
        \fi

        \expandafter\ifstrequal\expandafter{\article}{der}{
            \def\cardcolor{green}
        }{
            \expandafter\ifstrequal\expandafter{\article}{das}{
                \def\cardcolor{purple}
            }{
                \expandafter\ifstrequal\expandafter{\article}{die}{
                    \def\cardcolor{yellow}
                }{}
            }
        }

        \def\spellingcontent{
            \begin{tikzpicture}[baseline, every node/.style={spelling}]
                \node(t1)[text=\cardcolor, overlay] {\Large\article};
                \node(t2)[text=\cardcolor, overlay, right=of t1] {\Large\cardtitle,};
                \node(n1)[overlay, below=of t1] {\square[color=\cardcolor, border=true, scale=0.6]};
                \node(n2)[overlay] at ($(t2.center)!(n1.center)!(t2.center)$){\square[color=\cardcolor]};

                \expandafter\ifstrequal\expandafter{\genitive}{}{
                    \node(t3)[overlay, right=-0.5em of t2] {};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){};
                }{
                    \node(t3)[text=\cardcolor, overlay, minimum width=0.5cm, right=of t2] {\Large\genitive,};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){\olive[color=\cardcolor]};
                }

                \node(t4)[text=red, overlay, minimum width=0.5cm, right=of t3] {\Large\plural};
                \node(n4)[overlay] at ($(t4.center)!(n1.center)!(t4.center)$){\square[color=red]};
                % use fit to automatically center the content
                \node[fit=(t1) (t2) (t3) (t4) (n1) (n2) (n3) (n4), draw=none] (fitbox) {};

                \draw[-, shorten >=0, shorten <=0] (n1) -- (n2);

            \end{tikzpicture}%
        }
    }{}

    % verb
    \expandafter\ifstrequal\expandafter{\p}{v.}{
        % spelling
        \StrCount{\fspelling}{,}[\dc]
        \StrCut{\fspelling}{,}{\infinitive}{\pb}
        \def\cardtitle{\infinitive}
        \StrCut{\pb}{,}{\present}{\pc}
        \StrCut{\pc}{,}{\past}{\pd}
        \StrCut{\pd}{ }{\auxilary}{\pastparticle}

        \def\spellingcontent{
            \begin{tikzpicture}[baseline, every node/.style={spelling, verb}]
                \edef\conjugation{\present\past\auxilary}
                \expandafter\ifstrequal\expandafter{\conjugation}{}{
                    \node(t1)[text=black, font=\sffamily\bfseries, overlay] {\small\infinitive};
                    \node(n1)[overlay, below=1em of t1] {\cir[color=gray, scale=1]};
                }{
                    \node(t1)[text=black, font=\sffamily\bfseries, overlay] {\small\infinitive,};
                    \node(n1)[overlay, below=1em of t1] {\flower[color=gray, scale=1]};
                }

                \expandafter\ifstrequal\expandafter{\present}{}{
                    \node(t2)[overlay, right=-0.6cm of t1] {};
                    \node(n2)[overlay] at ($(t2.center)!(n1.center)!(t2.center)$){};
                }{
                    \node(t2)[text=black, overlay, right=of t1] {\small\present,};
                    \node(n2)[overlay] at ($(t2.center)!(n1.center)!(t2.center)$){\flower[color=gray, clock=simple]};
                }
                \expandafter\ifstrequal\expandafter{\past}{}{
                    \node(t3)[overlay, right=-0.6cm of t2] {};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){};
                }{
                    \node(t3)[text=black, overlay, right=of t2] {\small\past,};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){\flower[color=gray, clock=simple, rotate=true]};
                }
                \expandafter\ifstrequal\expandafter{\auxilary}{}{
                    \node(t4)[overlay, right=-0.6cm of t3] {};
                    \node(n4)[overlay] at ($(t4.center)!(n1.center)!(t4.center)$){};
                }{
                    \node(t4)[text=black, overlay, right=of t3] {\small\auxilary};
                    \node(n4)[overlay] at ($(t4.center)!(n1.center)!(t4.center)$){\flower[color=gray, fill=thin, clock=simple]};
                }
                \expandafter\ifstrequal\expandafter{\pastparticle}{}{
                    \node(t5)[overlay, right=-0.6cm of t4] {};
                    \node(n5)[overlay] at ($(t5.center)!(n1.center)!(t5.center)$){};
                }{
                    \node(t5)[text=black, overlay, right=of t4] {\small\pastparticle};
                    \node(n5)[overlay] at ($(t5.center)!(n1.center)!(t5.center)$){\flower[color=gray, pattern=vert, rotate=true]};
                }

                \node[fit=(t1) (t2) (t3) (t4) (t5) (n1) (n2) (n3) (n4), draw=none] (fitbox) {};
            \end{tikzpicture}%
        }
    }{}

    \expandafter\ifstrequal\expandafter{\p}{adj.}{
        \StrCut{\fspelling}{,}{\positive}{\pa}
        \def\cardtitle{\positive}
        \StrCut{\pa}{,}{\comparative}{\superlative}

        \def\spellingcontent{
            \begin{tikzpicture}[baseline, every node/.style={spelling, verb}]
                \edef\declension{\comparative\superlative}
                \expandafter\ifstrequal\expandafter{\declension}{}{
                    \node(t1)[text=black, font=\sffamily\bfseries, overlay] {\Large\positive};
                }{
                    \node(t1)[text=black, font=\sffamily\bfseries, overlay] {\Large\positive,};
                }
                \node(n1)[overlay, below=1em of t1] {\square[border=true, scale=0.8]};

                \expandafter\ifstrequal\expandafter{\comparative}{}{
                    \node(t2)[overlay, right=-0.6cm of t1] {};
                    \node(n2)[overlay] at ($(t2.center)!(n1.center)!(t2.center)$){};
                }{
                    \node(t2)[text=black, overlay, right=of t1] {\Large\comparative,};
                    \node(n2)[overlay] at ($(t2.center)!(n1.center)!(t2.center)$){\square[border=true, scale=0.8]};
                }
                \expandafter\ifstrequal\expandafter{\superlative}{}{
                    \node(t3)[overlay, right=-0.6cm of t2] {};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){};
                }{
                    \node(t3)[text=black, overlay, right=of t2] {\Large\superalative,};
                    \node(n3)[overlay] at ($(t3.center)!(n1.center)!(t3.center)$){\square[border=true, scale=0.8]};
                }

                \node[fit=(t1) (t2) (t3) (n1) (n2) (n3), draw=none] (fitbox) {};
            \end{tikzpicture}%
        }
    }{}

    \IfFileExists{\fpicture}{
        \def\picturepath{Blank.png}
    }
    {
        \def\picturepath{\fpicture}
    }

    \noindent
    \flashcard[color=\cardcolor, title=\cardtitle, foot=\ftranslation, spelling=\spellingcontent, picture=\picturepath, hint=\fhint]
    \newpage
}
