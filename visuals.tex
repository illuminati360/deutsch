% Square
\pgfkeys{
    /square/.is family, /square,
    default/.style = {color = {none}, border = {0.0}, fill = {full}},
    color/.store in = \squarecolor,
    border/.store in = \squareborder,
    fill/.store in = \squarefill
}

% must add `%` to prevent \NewDocumentCommand from adding extra spaces.
\NewDocumentCommand{\square}{o}{%
    \pgfkeys{/square, default, #1}%
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\a{2em}
        \def\l{0.1em}

        \pgfmathparse{\squareborder > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult

        % bottom half fill
        \expandafter\ifstrequal\expandafter{\squarefill}{half}{
            \def\fillcolor{none}
        }{
            \def\fillcolor{\squarecolor}
        }
        \draw[fill=\fillcolor, draw=\bordercolor, line width=\l] (0,0) rectangle (\a,\a);

        \expandafter\ifstrequal\expandafter{\squarefill}{half}{
            \expandafter\ifstrequal\expandafter{\squarecolor}{none}{}{
                \draw[fill=\squarecolor, draw=none] (\l/2,\l/2) rectangle (\a-\l/2,\a/2+\l/2);
            }
        }{}
    \end{tikzpicture}
}

\NewDocumentCommand{\olive}{o}{%
    \pgfkeys{/olive, default, #1}%
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \pgfmathsetmacro{\a}{2/1cm*1em}
        \pgfmathsetmacro{\b}{1.4/1cm*1em}
        \pgfmathparse{(\a*\a + \b*\b)/4/\b}
        \let\r\pgfmathresult
        \pgfmathparse{atan2(2*\r-\b, \a)}
        \let\x\pgfmathresult

        \pgfmathparse{\oliveborder > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult
        \draw[fill=\olivecolor, draw=\bordercolor, line width=0.1em]
        (\a/2,\r-\a/2) arc[start angle=\x, end angle=180-\x, x radius=\r, y radius=\r]
        (\a/2,\r-\a/2) arc[start angle=-\x, end angle=-180+\x, x radius=\r, y radius=\r];
    \end{tikzpicture}
}

% Flower
\pgfkeys{
    /flower/.is family, /flower,
    default/.style = {color = {white}, scale = {1.0}},
    color/.store in = \flowercolor,
    scale/.store in = \flowerscale,
}

\NewDocumentCommand{\flower}{o}{%
    \pgfkeys{/flower, default, #1}%
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\r{1em}
        \def\b{0.2em}
        \coordinate (o) at (0,0);

        \path[fill=\flowercolor, draw=none]
        ([shift={(0:\r)}]o)
        foreach \angle in {0,60,...,300} {
        let
        \p1 = ([shift={(\angle:\r)}]o), %A
        \p2 = ([shift={({\angle+90}:0.28*\r)}]\p1), %A1
        \p3 = ([shift={({\angle+30}:{\r-\b})}]o), %B
        \p4 = ([shift={({\angle+30-90}:{0.12*\r})}]\p3), %B1
        \p5 = ([shift={({\angle+30+90}:{0.12*\r})}]\p3), %B2
        \p6 = ([shift={({\angle+60}:{\r})}]o), %C
        \p7 = ([shift={({\angle-30}:{0.28*\r})}]\p6) %C1
        in
        -- (\p1) .. controls (\p2) and (\p4) ..
        (\p3) .. controls (\p5) and (\p7) .. (\p6)
        } -- cycle;
    \end{tikzpicture}
}

% Card
\pgfkeys{
    /card/.is family, /card,
    default/.style = {width = {17em}, height = {10em}},
    width/.store in = \cardwidth,
    height/.store in = \cardheight,
    label/.store in = \cardlabel,
}
\NewDocumentCommand{\card}{o m}{%
    \pgfkeys{/card, default, #1}%
    \begin{tikzpicture}
        \def\barh{4ex}
        \def\r{0.5em}

        % background
        \draw[rounded corners=0.5em, draw=black, line width=0.1em ]
        (0, 0) rectangle(\cardwidth, \cardheight);

        % top bar
        \draw[thick, fill=blue!10]
        (0, \cardheight-\barh) -- (0, \cardheight-\r) arc[start angle=180, end angle=90, radius=\r] --
        (\r, \cardheight) -- (\cardwidth-\r, \cardheight) arc[start angle=90, end angle=0, radius=\r] --
        (\cardwidth, \cardheight-\r) -- (\cardwidth, \cardheight-\barh) -- cycle;
        \node[anchor=west, font=\bfseries, xshift=0.5em, yshift=-0.2em] at (0, \cardheight-\barh/2) {\cardlabel};

        \node[anchor=north, text width=\cardwidth-1em] at (\cardwidth/2, \cardheight-\barh) {#2};
    \end{tikzpicture}
}
