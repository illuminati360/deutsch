\newcommand\dsq[1]{
    (-#1/2,-#1/2) rectangle (#1/2,#1/2);
}

% must add `%` to prevent \NewDocumentCommand from adding extra spaces.
\NewDocumentCommand{\square}{o}{%
    \pgfkeys{/square, default, #1}%
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\a{2em} %side length
        \def\l{0.06em} %line width

        \pgfmathparse{\squareborder > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult

        % only fill when \squarefill is full or large
        \def\fillcolor{none}
        \expandafter\ifstrequal\expandafter{\squarefill}{full}{
            \def\fillcolor{\squarecolor}
        }{}
        \expandafter\ifstrequal\expandafter{\squarefill}{large}{
            \def\fillcolor{\squarecolor}
        }{}

        % draw the outter square
        % \dsq{\fillcolor}{\bordercolor}{\l}{\a}
        \draw[fill=\fillcolor, draw=\bordercolor, line width=\l] \dsq{\a}

        % half fill
        \expandafter\ifstrequal\expandafter{\squarefill}{half}{
            \draw[fill=\squarecolor, draw=none] (-\a/2+\l/2,-\a/2\l/2) rectangle (-\a/2-\l/2,\l/2);
        }{}

        \def\ai{\a*0.8} % inner square side length
        % inner square if specified
        \expandafter\ifstrequal\expandafter{\squarefill}{large}{
            \draw[fill=none, draw=black, line width=\l] \dsq{\ai}
        }{}
        \expandafter\ifstrequal\expandafter{\squarefill}{small}{
            \draw[fill=none, draw=black, line width=\l] \dsq{\ai}
        }{}

        \expandafter\ifstrequal\expandafter{\squarefill}{middle}{
            \draw[fill=\squarecolor, draw=black, line width=\l] \dsq{\ai}
        }{}

        % patterns
        \expandafter\ifstrequal\expandafter{\squarepattern}{vert}{
            \draw[draw=black, line width=\l, shift={(-\a/2, -\a/2)}]
            (\a/3, 0) -- (\a/3, \a)
            (\a/3*2, 0) -- (\a/3*2, \a);
        }{}
        \expandafter\ifstrequal\expandafter{\squarepattern}{diag}{
            \draw[draw=black, line width=\l, shift={(-\a/2, -\a/2)}]
            (0, \a/2) -- (\a/2, 0)
            (0, \a) -- (\a, 0)
            (\a/2, \a) -- (\a, \a/2);
        }{}

        \ifdebug
            \draw[draw=red, line width=\l] (-\a/2,0) -- (\a/2,0);
            \draw[draw=red, line width=\l] (0,-\a/2) -- (0,\a/2);
        \fi

    \end{tikzpicture}
}

\newcommand\dol[3]{
    % #4 a, #5 x, #6 r
    % \draw[fill=#1, draw=#2, line width=#3]
    % (#4/2,0) arc[start angle=#5, end angle=180-#5, x radius=#6, y radius=#6]
    % (#4/2,0) arc[start angle=-#5, end angle=-180+#5, x radius=#6, y radius=#6];
    (#1/2,0) arc[start angle=#2, end angle=180-#2, x radius=#3, y radius=#3]

\newcommand\dfl[2]{
([shift={(0:#1)}]0,0)
foreach \angle in {0,60,...,300} {
let
\p1 = ([shift={(\angle:#1)}]0,0), %A
\p2 = ([shift={({\angle+90}:0.28*#1)}]\p1), %A1, control point
\p3 = ([shift={({\angle+30}:{#1-#2})}]0,0), %B
\p4 = ([shift={({\angle+30-90}:{0.12*#1})}]\p3), %B1, control point
\p5 = ([shift={({\angle+30+90}:{0.12*#1})}]\p3), %B2, control point
\p6 = ([shift={({\angle+60}:{#1})}]0,0), %C
\p7 = ([shift={({\angle-30}:{0.28*#1})}]\p6) %C1
in
-- (\p1) .. controls (\p2) and (\p4) ..
(\p3) .. controls (\p5) and (\p7) .. (\p6)
} -- cycle;
}
