\NewDocumentCommand{\square}{m O{0.0}}{
    % #1 = fill color, #2 = line width
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\a{2em}

        \pgfmathparse{#2 > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult
        \draw[fill=#1, draw=\bordercolor, line width=0.1] (0,0) rectangle (\a,\a);
    \end{tikzpicture}
}

\NewDocumentCommand{\diam}{m O{0.0}}{
    % #1 = fill color, #2 = line width
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\a{2.5em}
        \def\b{2em}

        \pgfmathparse{#2 > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult
        \draw[fill=#1, draw=\bordercolor, line width=0.1] (0,\b/2) -- (\a/2,0) -- (\a,\b/2) -- (\a/2,\b) -- cycle;
    \end{tikzpicture}
}

\NewDocumentCommand{\olive}{m O{0.0}}{
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \pgfmathsetmacro{\a}{2/1cm*1em}
        \pgfmathsetmacro{\b}{1.4/1cm*1em}
        \pgfmathparse{(\a*\a + \b*\b)/4/\b}
        \let\r\pgfmathresult
        \pgfmathparse{atan2(2*\r-\b, \a)}
        \let\x\pgfmathresult

        \pgfmathparse{#2 > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult
        \draw[fill=#1, draw=\bordercolor, line width=0.1]
        (\a/2,\r-\a/2) arc[start angle=\x, end angle=180-\x, x radius=\r, y radius=\r]
        (\a/2,\r-\a/2) arc[start angle=-\x, end angle=-180+\x, x radius=\r, y radius=\r];
    \end{tikzpicture}
}

\NewDocumentCommand{\tri}{m O{0.0}}{
    \begin{tikzpicture}[baseline={([yshift=-0.5em]current bounding box.center)}]
        \def\a{2em}
        \def\b{1.5em}

        \pgfmathparse{#2 > 0 ? "black" : "none"}
        \let\bordercolor\pgfmathresult
        \draw[fill=#1, draw=\bordercolor, line width=0.1]
        (-\a/2, 0) -- (\a/2, \b/2) -- (\a/2, -\b/2) -- cycle;
    \end{tikzpicture}
}
